# 用户建议反馈 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 首页「建议反馈」入口（无数字红点）+ 列表顶矮输入提交 + 列表卡片（折叠渐隐、已读、分页缓存）+ 运营脚本回复 + 订阅消息（点击进首页再跳列表）。

**Architecture:** 云函数 `feedback`（create / listMine / markRead / hasUnread / reply）；`feedbacks` 用 `unreadForUser` + `sortKey`；`users.feedbackListEpoch` 失效列表缓存；仅 `pages/feedback/list`（顶部常驻写作区）；`reply` 时若 `subscribeReady` 发订阅消息，落地 `pages/index/index?openFeedback=1`。

**Tech Stack:** 微信小程序原生；云开发 Node；`node --test`；`FEEDBACK_ADMIN_KEY` + `FEEDBACK_SUBSCRIBE_TMPL_ID`。

**Specs:**

## Global Constraints

- 入口仅首页 link-row；不改 `mine`。  
- 红点无数字；与 `profileGapCount` 分离。  
- 正文 ≤100 字；运营回复不截断。  
- 点已读只消该条红点、**不重排**。  
- 列表缓存 TTL 7 天；运营回复 bump epoch。  
- 订阅未授权时站内路径仍完整；不做详情页；不进 recommend reject。

```js
'use strict'

const CONTENT_MAX = 100
const UNREAD_SORT_BASE = 2e15
const CATEGORY_LABELS = {
  recipe: '菜谱',
  ingredient: '食材',
  feature: '功能建议',
  other: '其他'
}

function toMs(v) {
  if (v == null) return null
  if (typeof v === 'number') return v
  if (v instanceof Date) return v.getTime()
  if (typeof v === 'string') {
    const t = Date.parse(v)
    return Number.isFinite(t) ? t : null
  }
  if (typeof v.valueOf === 'function') {
    const n = Number(v.valueOf())
    return Number.isFinite(n) ? n : null
  }
  return null
}

function validateCreate(input) {
  const category = input && input.category
  if (!CATEGORY_LABELS[category]) {
    return { ok: false, error: 'BAD_CATEGORY' }
  }
  const content = String((input && input.content) || '').trim()
  if (!content) return { ok: false, error: 'EMPTY_CONTENT' }
  if (content.length > CONTENT_MAX) return { ok: false, error: 'CONTENT_TOO_LONG' }
  return { ok: true, category, content }
}

function isUnread(doc) {
  const text = doc && doc.replyText
  if (!text || !String(text).trim()) return false
  const repliedAt = toMs(doc.repliedAt)
  const readAt = toMs(doc.userReadAt)
  if (readAt == null) return true
  if (repliedAt == null) return true
  return readAt < repliedAt
}

function computeSortKey(input) {
  const createdAtMs = Number(input && input.createdAtMs) || 0
  const repliedAtMs = Number(input && input.repliedAtMs) || 0
  const base = Math.max(createdAtMs, repliedAtMs)
  if (input && input.unreadForUser) return UNREAD_SORT_BASE + base
  return base
}

function shapeFeedback(doc) {
  if (!doc) return null
  const unread =
    typeof doc.unreadForUser === 'boolean' ? doc.unreadForUser : isUnread(doc)
  return {
    id: doc._id || doc.id,
    category: doc.category,
    categoryLabel: CATEGORY_LABELS[doc.category] || doc.category,
    content: doc.content || '',
    createdAt: toMs(doc.createdAt),
    replyText: doc.replyText ? String(doc.replyText) : null,
    repliedAt: toMs(doc.repliedAt),
    unread: !!unread
  }
}

module.exports = {
  CONTENT_MAX,
  UNREAD_SORT_BASE,
  CATEGORY_LABELS,
  toMs,
  validateCreate,
  isUnread,
  computeSortKey,
  shapeFeedback
}
```

```js
'use strict'
const cloud = require('wx-server-sdk')
const {
  validateCreate,
  computeSortKey,
  shapeFeedback,
  toMs,
  CONTENT_MAX
} = require('./feedbackLogic.js')

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()
const _ = db.command

async function resolveUid(openid) {
  const idRes = await db
    .collection('user_identities')
    .where({ provider: 'wechat_mp', providerUserId: openid })
    .limit(1)
    .get()
  return (idRes.data[0] && idRes.data[0].uid) || null
}

async function getUserDoc(uid) {
  const res = await db.collection('users').where({ uid }).limit(1).get()
  return res.data[0] || null
}

async function handleCreate(event, uid) {
  const v = validateCreate(event)
  if (!v.ok) return { ok: false, error: v.error }

  const recent = await db
    .collection('feedbacks')
    .where({ uid })
    .orderBy('createdAt', 'desc')
    .limit(1)
    .get()
  const last = recent.data[0]
  const lastMs = toMs(last && last.createdAt)
  if (lastMs != null && Date.now() - lastMs < 60 * 1000) {
    return { ok: false, error: 'RATE_LIMITED' }
  }

  const now = db.serverDate()
  const createdAtMs = Date.now()
  const subscribeReady = !!(event && event.subscribeReady)
  const addRes = await db.collection('feedbacks').add({
    data: {
      uid,
      category: v.category,
      content: v.content,
      createdAt: now,
      replyText: null,
      repliedAt: null,
      repliedBy: null,
      userReadAt: null,
      unreadForUser: false,
      subscribeReady,
      sortKey: computeSortKey({
        unreadForUser: false,
        createdAtMs,
        repliedAtMs: 0
      })
    }
  })
  const item = shapeFeedback({
    _id: addRes._id,
    category: v.category,
    content: v.content,
    createdAt: createdAtMs,
    replyText: null,
    repliedAt: null,
    unreadForUser: false
  })
  return { ok: true, item }
}

async function handleHasUnread(uid) {
  const user = await getUserDoc(uid)
  const epoch = (user && user.feedbackListEpoch) || 0
  const res = await db
    .collection('feedbacks')
    .where({ uid, unreadForUser: true })
    .limit(1)
    .get()
  return {
    ok: true,
    hasUnread: !!(res.data && res.data.length),
    feedbackListEpoch: epoch
  }
}

async function handleListMine(event, uid) {
  const limit = Math.min(20, Math.max(1, Number(event.limit) || 5))
  const cursor = event.cursor || null
  const user = await getUserDoc(uid)
  const epoch = (user && user.feedbackListEpoch) || 0

  // 多取若干便于内存 cursor（若云库 where+orderBy 直接支持则优先服务端分页）
  const res = await db
    .collection('feedbacks')
    .where({ uid })
    .orderBy('sortKey', 'desc')
    .limit(100)
    .get()

  let rows = res.data || []
  if (cursor && cursor.sortKey != null) {
    const cKey = Number(cursor.sortKey)
    const cId = String(cursor.id || '')
    rows = rows.filter((row) => {
      const k = Number(row.sortKey) || 0
      if (k < cKey) return true
      if (k > cKey) return false
      return String(row._id) < cId
    })
  }
  const page = rows.slice(0, limit)
  const items = page.map(shapeFeedback)
  let nextCursor = null
  if (page.length === limit && rows.length > limit) {
    const last = page[page.length - 1]
    nextCursor = {
      sortKey: Number(last.sortKey) || 0,
      id: last._id
    }
  }
  const unreadRes = await db
    .collection('feedbacks')
    .where({ uid, unreadForUser: true })
    .limit(1)
    .get()
  return {
    ok: true,
    items,
    nextCursor,
    feedbackListEpoch: epoch,
    hasUnread: !!(unreadRes.data && unreadRes.data.length)
  }
}

async function handleMarkRead(event, uid) {
  const id = event && event.id
  if (!id) return { ok: false, error: 'ID_REQUIRED' }
  const docRes = await db.collection('feedbacks').doc(id).get()
  const doc = docRes.data
  if (!doc || doc.uid !== uid) return { ok: false, error: 'NOT_FOUND' }
  if (!doc.replyText) return { ok: true, item: shapeFeedback(doc) }

  const createdAtMs = toMs(doc.createdAt) || Date.now()
  const repliedAtMs = toMs(doc.repliedAt) || createdAtMs
  const sortKey = computeSortKey({
    unreadForUser: false,
    createdAtMs,
    repliedAtMs
  })
  await db.collection('feedbacks').doc(id).update({
    data: {
      userReadAt: db.serverDate(),
      unreadForUser: false,
      sortKey
    }
  })
  return {
    ok: true,
    item: shapeFeedback(
      Object.assign({}, doc, {
        userReadAt: Date.now(),
        unreadForUser: false,
        sortKey
      })
    )
  }
}

async function handleReply(event) {
  const key = process.env.FEEDBACK_ADMIN_KEY || ''
  if (!key || event.adminKey !== key) {
    return { ok: false, error: 'FORBIDDEN' }
  }
  const id = event.id
  const replyText = String((event && event.replyText) || '').trim()
  if (!id || !replyText) return { ok: false, error: 'BAD_ARGS' }
  // replyText 不按 CONTENT_MAX 截断
  void CONTENT_MAX

  const docRes = await db.collection('feedbacks').doc(id).get()
  const doc = docRes.data
  if (!doc) return { ok: false, error: 'NOT_FOUND' }

  const repliedAtMs = Date.now()
  const createdAtMs = toMs(doc.createdAt) || repliedAtMs
  const sortKey = computeSortKey({
    unreadForUser: true,
    createdAtMs,
    repliedAtMs
  })
  await db.collection('feedbacks').doc(id).update({
    data: {
      replyText,
      repliedAt: db.serverDate(),
      repliedBy: String(event.repliedBy || 'admin'),
      userReadAt: null,
      unreadForUser: true,
      sortKey
    }
  })
  const user = await getUserDoc(doc.uid)
  if (user && user._id) {
    await db
      .collection('users')
      .doc(user._id)
      .update({
        data: {
          feedbackListEpoch: _.inc(1),
          updatedAt: db.serverDate()
        }
      })
  }

  // 订阅消息：失败只打日志
  if (doc.subscribeReady) {
    try {
      const idRes = await db
        .collection('user_identities')
        .where({ uid: doc.uid, provider: 'wechat_mp' })
        .limit(1)
        .get()
      const openid = idRes.data[0] && idRes.data[0].providerUserId
      const tmplId = process.env.FEEDBACK_SUBSCRIBE_TMPL_ID || ''
      if (openid && tmplId) {
        await cloud.openapi.subscribeMessage.send({
          touser: openid,
          templateId: tmplId,
          page: 'pages/index/index?openFeedback=1',
          data: {
            // 字段名以微信后台模板为准，实现时替换；示意：
            // thing1: { value: CATEGORY_LABELS[doc.category] || '建议' },
            // thing2: { value: replyText.slice(0, 20) }
          }
        })
      }
    } catch (err) {
      console.error('subscribeMessage.send failed', err)
    }
  }
  return { ok: true }
}

exports.main = async (event) => {
  const wxContext = cloud.getWXContext()
  const action = (event && event.action) || ''
  if (action === 'reply') return handleReply(event)

  const openid = wxContext.OPENID
  if (!openid) return { ok: false, error: 'NO_OPENID' }
  const uid = await resolveUid(openid)
  if (!uid) return { ok: false, error: 'NO_USER' }

  if (action === 'create') return handleCreate(event, uid)
  if (action === 'listMine') return handleListMine(event, uid)
  if (action === 'markRead') return handleMarkRead(event, uid)
  if (action === 'hasUnread') return handleHasUnread(uid)
  return { ok: false, error: 'UNKNOWN_ACTION' }
}
```

- [ ] **Step 3: 提交（先订阅再 create）**

```js
onSubmit() {
  if (this.data.submitting) return
  const content = String(this.data.draft || '').trim()
  if (!content) {
    wx.showToast({ title: '请填写内容', icon: 'none' })
    return
  }
  if (content.length > 100) {
    wx.showToast({ title: '最多100字', icon: 'none' })
    return
  }
  this.setData({ submitting: true })
  const tmplId = /* 与后台一致，可写在 utils/subscribe.js 常量 */ ''
  const afterAuth = (subscribeReady) => {
    track('feedback_subscribe_result', { accepted: !!subscribeReady })
    return callCloud('feedback', {
      action: 'create',
      category: this.data.category,
      content,
      subscribeReady: !!subscribeReady
    })
      .then((res) => {
        const item = res && res.item
        if (!item) throw new Error('CREATE_FAILED')
        track('feedback_compose_submit', {
          category: this.data.category,
          contentLen: content.length,
          subscribeReady: !!subscribeReady
        })
        const items = [item].concat(this.data.items || [])
        this.setData({ draft: '', draftLen: 0, items, submitting: false })
        this._persistCache()
        wx.showModal({
          title: '',
          content: '感谢您的反馈',
          showCancel: false
        })
      })
      .catch((err) => {
        this.setData({ submitting: false })
        const code = err && (err.error || err.message)
        wx.showToast({
          title: code === 'RATE_LIMITED' ? '提交太频繁，稍后再试' : '提交失败',
          icon: 'none'
        })
      })
  }
  if (!tmplId) {
    afterAuth(false)
    return
  }
  wx.requestSubscribeMessage({
    tmplIds: [tmplId],
    success: (r) => afterAuth(r[tmplId] === 'accept'),
    fail: () => afterAuth(false)
  })
}
```

`tmplId` 用小程序端可配置常量（非 adminKey）；未配置时跳过授权直接 create。
